TL‑Verilog Examples

1-bit Inverter

// TL‑Verilog: 1-bit inverter
`default_nettype none

module inverter(input logic a, output logic y);
  always_comb y = ~a;
endmodule

1-bit Half Adder

// TL‑Verilog: half adder
module half_adder(input logic a, b, output logic sum, cout);
  assign sum = a ^ b;
  assign cout = a & b;
endmodule

1-bit Full Adder

// TL‑Verilog: full adder
module full_adder(input logic a, b, cin, output logic sum, cout);
  assign {cout, sum} = a + b + cin;
endmodule
